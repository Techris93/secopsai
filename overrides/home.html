Quick start

bash 
# Install and initialize
curl -fsSL https://secopsai.dev/install.sh | bash
cd ~/secopsai
source .venv/bin/activate
secopsai refresh
secopsai refresh --platform macos,openclaw,hermes
secopsai correlate
secopsai triage orchestrate --search-root ~/secopsai
